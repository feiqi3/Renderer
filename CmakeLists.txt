cmake_minimum_required(VERSION 3.10)
project(Renderer)

# Set C++ standard to C++17
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

# Optionally set the build type to Release if not specified
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Gather all source files from the src folder
file(GLOB_RECURSE SOURCES "src/*.cpp" "include/*.h")

# Create the Renderer executable
add_executable(Renderer ${SOURCES})

message("Sources: ${SOURCES}")

add_subdirectory(3rd)

# Add include directory for header files
target_include_directories(Renderer PUBLIC include)

# Link libraries if required

target_include_directories(Renderer PRIVATE 3rd/vma)
target_include_directories(Renderer PRIVATE 3rd/volk)
target_link_libraries(Renderer PRIVATE spirv-reflect-static)
set_target_properties(spirv-reflect-static PROPERTIES FOLDER "ThirdParty/")
target_link_libraries(Renderer PRIVATE shaderc)
set_target_properties(shaderc PROPERTIES FOLDER "ThirdParty/")
target_link_libraries(Renderer PRIVATE glfw)
set_target_properties(glfw update_mappings  PROPERTIES FOLDER "ThirdParty/glfw")
target_link_libraries(Renderer PRIVATE volk::volk)
set_target_properties(volk PROPERTIES FOLDER "ThirdParty/")
target_link_libraries(Renderer PUBLIC glm)
set_target_properties(glm PROPERTIES FOLDER "ThirdParty/")
target_compile_definitions(Renderer PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE)
